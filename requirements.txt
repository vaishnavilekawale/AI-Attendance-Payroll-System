# =============================================================================
# requirements.txt — AI Attendance & Payroll System (runtime dependencies)
# =============================================================================
# Target interpreter: Python 3.10.x (3.10.11 recommended). See README.md
# Section 2.1 for why 3.10 specifically — TensorFlow 2.15, mediapipe 0.10.21,
# and DeepFace's transitive dependencies all publish official Windows wheels
# for 3.9–3.11, and 3.10 is the safest intersection. Do NOT use Python 3.12+.
#
# Install into a clean virtual environment:
#     python -m venv venv
#     venv\Scripts\activate
#     python -m pip install --upgrade pip
#     pip install -r requirements.txt
#
# Do NOT `pip install` on top of an old venv that may still have plain
# `opencv-python` installed alongside `opencv-contrib-python` — the two
# packages both provide the `cv2` module and having both installed causes
# unpredictable behavior. If in doubt:
#     pip uninstall -y opencv-python opencv-contrib-python
#     pip install -r requirements.txt
#
# Build/packaging-only tools (PyInstaller, pyinstaller-hooks-contrib) are
# intentionally NOT in this file — see requirements-packaging.txt. Customers
# running the packaged .exe never need those; only your build machine does.
#
# Verify the install succeeded before touching PyInstaller:
#     python -c "import cv2, tensorflow, mediapipe, deepface; print('OK')"
# =============================================================================

# Numerical / data handling
# ---------------------------------------------------------------------------
# Pinned <2.0 deliberately: TensorFlow 2.15 and mediapipe 0.10.21 (below)
# were both built against NumPy 1.x's ABI and are NOT compatible with
# NumPy 2.x - installing numpy>=2 breaks face recognition at import time
# with a binary-incompatibility error, not a clean ImportError.
numpy==1.26.4
pandas==2.2.3

# ---------------------------------------------------------------------------
# Computer vision / face recognition stack
# ---------------------------------------------------------------------------
# Plain `opencv-python` must NEVER also be installed alongside this - both
# packages provide the same `cv2` import and having both present causes
# whichever installed last to silently "win", which is a nightmare to
# debug. Only opencv-contrib-python belongs in this file.
opencv-contrib-python==4.8.1.78
mediapipe==0.10.21
# TensorFlow 2.15 is the pinned major/minor for Windows wheel availability
# with mediapipe 0.10.21 and DeepFace (see README.md Section 2.1). tf-keras
# provides the Keras 2 API that DeepFace's model-loading code expects; it
# must be installed alongside tensorflow==2.15 (which still defaults to
# Keras 2 natively, but pinning tf-keras explicitly avoids ambiguity if pip
# ever resolves a newer, Keras-3-only TensorFlow as a transitive dependency
# of something else).
tensorflow==2.15.0
tf-keras==2.15.1
deepface==0.0.93
gdown==5.2.0
